CREATE TABLE payment (
    payment_id INT AUTO_INCREMENT PRIMARY KEY,
    firstname VARCHAR(50),
    lastname VARCHAR(50),
    email VARCHAR(100),
    method ENUM('card-payment', 'cash-payment'),
    creditCardNum VARCHAR(16),
    date VARCHAR(5),
    cvv VARCHAR(3),
    receipt VARCHAR(255),
    totalPrice DECIMAL(10, 2),
    confirmed BOOLEAN DEFAULT 0
);
