CREATE TABLE Reservation(
    ReservationID INT PRIMARY KEY,
    CID INT,
    BranchID INT,
    Destination VARCHAR(50),
    NoOfAdults INT,
    NoOfChildren INT,
    PickupTime TIME,
    ReturnTime TIME,
    PickupDate DATE,
    ReturnDate DATE,
    FOREIGN KEY(CID) REFERENCES customer(CID),
    FOREIGN KEY(BranchID) REFERENCES Branch(BranchID)
    );
